module SumadorCompletoUnBit(
    input bitEntradaA,
    input bitEntradaB,
    input acarreoEntrada,
    output bitResultadoSuma,
    output acarreoSalida
);

    // Señales internas
    wire resultadoXorEntradas;
    wire acarreoGeneradoPorEntradas;
    wire acarreoGeneradoPorAcarreo;

    // Cálculo de la suma
    assign resultadoXorEntradas = bitEntradaA ^ bitEntradaB;
    assign bitResultadoSuma = resultadoXorEntradas ^ acarreoEntrada;

    // Cálculo de los acarreos
    assign acarreoGeneradoPorEntradas =
           bitEntradaA & bitEntradaB;

    assign acarreoGeneradoPorAcarreo =
           acarreoEntrada & resultadoXorEntradas;

    // Acarreo final
    assign acarreoSalida =
           acarreoGeneradoPorEntradas |
           acarreoGeneradoPorAcarreo;

endmodule
